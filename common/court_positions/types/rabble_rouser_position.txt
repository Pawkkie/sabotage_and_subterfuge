				}
				else_if = {
					limit = {
						has_trait_xp = { trait = lifestyle_reveler value <= 99 }
					}
					value = 20
				}
				else_if = {
					limit = {
						has_trait_xp = { trait = lifestyle_reveler value > 99 }
					}
					value = 35
				}
				desc = court_position_reveler_trait
			}

		# Traveler
		if = {
			limit = { has_trait = lifestyle_traveler }
			add = {
				if = {
					limit = {
						OR = {
							has_trait_xp = { trait = lifestyle_traveler track = travel value <= 49 }

					value = 10
				}

						OR = {

					value = 20

						OR = {

			add = {
				value = 5

					limit = { has_trait = education_diplomacy_2 }

					limit = { has_trait = education_diplomacy_4 }

				}

					limit = { has_trait = education_diplomacy_5 }

				}
				desc = trait_education_diplomacy

		}
		if = {
			limit = {
				OR = {
					has_trait = education_intrigue_1
					has_trait = education_intrigue_2
					has_trait = education_intrigue_3
					has_trait = education_intrigue_4
					has_trait = education_intrigue_5

				value = 5
				if = {
					limit = { has_trait = education_intrigue_2 }
					add = 5
				}
				else_if = {
					limit = { has_trait = education_intrigue_3 }
					add = 10

					limit = { has_trait = education_intrigue_4 }
					add = 15
				}
				else_if = {
					limit = { has_trait = education_intrigue_5 }
					add = 20
				}
				desc = trait_education_intrigue
			}
		}
		# Seneschal
		if = {
			limit = {
				liege.court_position:seneschal_court_position ?= {
					has_variable = performing_improve_court
				}
			}
			add = {
				value = liege.court_position:seneschal_court_position.seneschal_improve_court_value
				desc = seneschal_improve_court_aptitude_desc
			}
		}
	}

	is_shown = {
		OR = {
			is_landed_or_landless_administrative = yes
			government_has_flag = government_is_nomadic
		}
		trigger_if = {
			limit = {
				is_ai = yes
				highest_held_title_tier <= tier_county
			}
			always = no
		}
	}

	valid_position = {
		is_landed_or_landless_administrative = yes
		highest_held_title_tier >= tier_county
		OR = {
			has_government = clan_government
			has_government = tribal_government
			custom_description = {
				text = already_established_criminal_underground
				house = { has_house_modifier = sas_establish_criminal_underground_house_modifier }
			}
		}
	}

	is_shown_character = {
		scope:employee = {
			is_imprisoned = no
		}
	}

	valid_character = {
		scope:employee = {
			rabble_rouser_validity_trigger = { EMPLOYER = scope:liege }
		}
	}

	revoke_cost = {
		prestige =  {
			value = major_court_position_prestige_revoke_cost
			# Set to 0 via obligation hook
			if = {
				limit = {
					court_position_obligation_hook_salary_trigger = {
						POS = rabble_rouser
						LIEGE = root
					}
				}
				multiply = 0
			}
			# Avoid AI prestige issues
			if = {
				limit = {
					exists = this
					is_ai = yes
				}
				multiply = 0
			}
			# Special case for events
			if = {
				limit = {
					exists = this
					has_character_flag = free_revoke_cost
				}
				multiply = 0
			}
			#Lowering for LAAMPS, to match Camp Officer revoke cost
			if = {
				limit = {
					exists = this
					government_has_flag = government_is_landless_adventurer
				}
				value = minor_court_position_prestige_revoke_cost
			}
		}
	}

	salary = {
		round = no
		gold = {
			value = medium_court_position_salary # Update X_salary_tooltip if script_value is changed
			# Set to 0 with Dynasty Perk Inner Circle
			if = {
				limit = { court_position_inner_circle_salary_trigger = { POS = rabble_rouser } }
				multiply = {
					value = 0
					desc = inner_circle_salary_mod
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			# Halved with obligation hook
			else_if = {
				limit = {
					court_position_obligation_hook_salary_trigger = {
						POS = rabble_rouser
						LIEGE = scope:liege
					}
				}
				multiply = {
					value = 0.5
					desc = obligation_hook_salary
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			# Halved salary
			if = {
				limit = {
					court_position_halved_salary_trigger = { POS = rabble_rouser }
				}
				multiply = {
					value = 0.5
					desc = court_position_halved_salary
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}
			}
			# Increased salary
			if = {
				limit = {
					court_position_increased_salary_trigger = { POS = rabble_rouser }
				}
				multiply = {
					value = 1.5
					desc = court_position_increased_salary
					format = "BASE_VALUE_FORMAT_DECIMALS_MINUS_POSITIVE"
				}

			#You're working with LAAMPS now

				limit = {

				multiply = {
					value = 0
					desc = court_position_laamps_free

				}
			}

			value = medium_court_position_prestige_salary # Update X_salary_tooltip if script_value is changed
			# Set to 0 with Dynasty Perk Inner Circle
			if = {

					value = 0

					value = 0.5

					value = 0

	is_powerful_agent = yes

	scaling_employer_modifiers = {
		terrible = {
			character_capital_county_monthly_control_add = -0.2
			county_opinion_add = -5

		poor = {
			character_capital_county_monthly_control_add = -0.1
			county_opinion_add = -2

			character_capital_county_monthly_control_add = 0.1

			character_capital_county_monthly_control_add = 0.3
			county_opinion_add = 7

		excellent = {
			character_capital_county_monthly_control_add = 0.5
			county_opinion_add = 10

	custom_employer_modifier_description = rabble_rouser_custom_effect_description

		intrigue = 1
		diplomacy = 1
		general_opinion = 10
		lifestyle_reveler_xp_gain_mult = 0.25
		monthly_diplomacy_lifestyle_xp_gain_mult = 0.1
		monthly_intrigue_lifestyle_xp_gain_mult =  0.1
	}

	on_court_position_received = {
		basic_gained_court_position_effect = yes
		scope:liege = {
			set_variable = {
				name = rabble_rouser_is_rousing_rabble
				value = no
			}
			set_variable = {
				name = rabble_rouser_is_provoking_vassals
				value = no
			}
			set_variable = {
				name = rabble_rouser_is_pacifying_the_masses
				value = no
			}
		}
	}
	on_court_position_revoked = {
		basic_revoked_court_position_effect = yes
		scope:liege = { 
			remove_variable = rabble_rouser_is_rousing_rabble
			remove_variable = rabble_rouser_is_pacifying_the_masses
			remove_variable = rabble_rouser_is_provoking_vassals
		}
	}
	on_court_position_invalidated = {
		basic_invalidated_court_position_effect = yes
		scope:liege = { 
			remove_variable = rabble_rouser_is_rousing_rabble
			remove_variable = rabble_rouser_is_pacifying_the_masses
			remove_variable = rabble_rouser_is_provoking_vassals
		}
	}
	on_court_position_vacated = {
		court_position_vacated_event_effect = {
			COURT_POSITION_TYPE = rabble_rouser_court_position
		}
		scope:liege = { 
			remove_variable = rabble_rouser_is_rousing_rabble
			remove_variable = rabble_rouser_is_pacifying_the_masses
			remove_variable = rabble_rouser_is_provoking_vassals
		}
	}

	candidate_score = {
		value = 50
		add = scope:base_value
		add = court_position_candidate_score_base_value
		scope:liege = {
			# More likely to hire if a Schemer
			if = {
				limit = {
					has_trait = schemer
				}
				add = 25
			}
		}
		scope:liege = {
			if = {
				limit = {
					OR = {
						has_trait = gregarious
						has_trait = lifestyle_reveler
					}
				}
				add = 25
			}
		}
		# More likely to hire with low capital control
		scope:liege.capital_county = {
			if = {
				limit = {
					county_control < full_county_control
				}
				add = 50
			}
		}
		if = {
			limit = {
				scope:employee_aptitude > 2
			}
			add = court_position_candidate_aptitude_value
		}
		if = { # To stop petitioned positions immediately being fired by the AI
			limit = {
				court_position_petition_ai_weight_trigger = { ROLE = smuggler }
			}
			add = 1000
		}
		add = court_position_debt_considerations_value
	}
