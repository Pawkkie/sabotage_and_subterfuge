femme_fatale_court_position = { 
	sort_order = 255
	max_available_positions = 1
	skill = intrigue
	minimum_rank = county

	court_position_asset = {
		animation = dancing
		background = "gfx/interface/illustrations/event_scenes/fp2_corridor_night.dds"
	}

	opinion = {
		value = regular_court_position_opinion
	}

	aptitude_level_breakpoints = { 20 40 60 80 }

	aptitude = {
		value = 10
		add = court_position_aptitude_boon_voyager_trait
		add = {
			value = intrigue
			multiply = 2
			max = 50
			desc = court_position_skill_intrigue
		}
		add = {
			value = diplomacy
			multiply = 1
			max = 50
			desc = court_position_skill_diplomacy
		}
		# Negative
		if = {
			limit = { has_trait = craven }
			add = { value = -15 desc = court_position_craven_trait }
		}
		if = {
			limit = { has_trait = lazy }
			add = { value = -15 desc = court_position_lazy_trait }
		}
		if = {
			limit = { has_trait = shy }
			add = { value = -30 desc = court_position_shy_trait }
		}		
		if = {
			limit = { has_trait = reclusive }
			add = { value = -15 desc = court_position_reclusive_trait }
		}
		if = {
			limit = { has_trait = infirm }
			add = { value = -75 desc = court_position_infirm_trait }
		}
		if = {
			limit = { has_trait = disloyal }
			add = { value = -15 desc = court_position_disloyal_trait }
		}
		# Positive
		if = {
			limit = { has_trait = lustful }
			add = { value = 25 desc = court_position_lustful_trait }
		}
		if = {
			limit = { has_trait = ambitious }
			add = { value = 15 desc = court_position_ambitious_trait }
		}
		if = {
			limit = { has_trait = vengeful }
			add = { value = 20 desc = court_position_ambitious_trait }
		}
		if = {
			limit = { has_trait = deceitful }
			add = { value = 25 desc = court_position_deceitful_trait }
		}
		if = {
			limit = { has_trait = fickle }
			add = { value = 15 desc = court_position_fickle_trait }
		}
		if = {
			limit = { has_trait = beauty_good_1 }
			add = { value = 15 desc = court_position_beauty_good_1_trait }
		}
		if = {
			limit = { has_trait = beauty_good_2 }
			add = { value = 25 desc = court_position_beauty_good_2_trait }
		}
		if = {
			limit = { has_trait = beauty_good_3 }
			add = { value = 35 desc = court_position_beauty_good_3_trait }
		}
		if = {
			limit = { has_trait = diplomat }
			add = { value = 20 desc = court_position_diplomat_trait}
		}
		if = {
			limit = { has_trait = schemer }
			add = { value = 25 desc = court_position_schemer_trait }
		}
		if = {
			limit = { has_trait = seducer }
			add = { value = 35 desc = court_position_seducer_trait }
		}
		if = {
			limit = { has_trait = adulterer }
			add = { value = 20 desc = court_position_adulterer_trait }
		}
		# Education
		if = {
			limit = {
				OR = {
					has_trait = education_diplomacy_1
					has_trait = education_diplomacy_2
					has_trait = education_diplomacy_3
					has_trait = education_diplomacy_4
					has_trait = education_diplomacy_5
				}
			}
			add = {
				value = 5
				if = {
					limit = { has_trait = education_diplomacy_2 }
					add = 5
				}
				else_if = {
					limit = { has_trait = education_diplomacy_3 }
					add = 10
				}
				else_if = {
					limit = { has_trait = education_diplomacy_4 }
					add = 15
				}
				else_if = {
					limit = { has_trait = education_diplomacy_5 }
					add = 20
				}
				desc = trait_education_diplomacy
			}
		}
		if = {
			limit = {
				OR = {
					has_trait = education_intrigue_1
					has_trait = education_intrigue_2
					has_trait = education_intrigue_3
					has_trait = education_intrigue_4
					has_trait = education_intrigue_5
				}
			}
			add = {
				value = 5
				if = {
					limit = { has_trait = education_intrigue_2 }
					add = 5
				}
				else_if = {
					limit = { has_trait = education_intrigue_3 }
					add = 10
				}
				else_if = {
					limit = { has_trait = education_intrigue_4 }
					add = 15
				}
				else_if = {
					limit = { has_trait = education_intrigue_5 }
					add = 20
				}
				desc = trait_education_intrigue
			}
		}
		add = court_position_aptitude_eunuch_tradition_value
		add = court_position_aptitude_traveling_friends_value

		add = court_position_aptitude_family_business_value
		add = court_position_aptitude_high_penalty_value
		# Bonus from court.8311
		if = {
			limit = { var:bonus_aptitude_role ?= flag:femme_fatale }
			add = {
				value = cp_bonus_aptitude_value
				desc = cp_bonus_aptitude
			}
		}
		# Seneschal
		if = {
			limit = {
				liege.court_position:seneschal_court_position ?= {
					has_variable = performing_improve_court
				}
			}
			add = {
				value = liege.court_position:seneschal_court_position.seneschal_improve_court_value
				desc = seneschal_improve_court_aptitude_desc
			}
		}
		# Yeke Jarquchi
		if = {
			limit = {
				liege.court_position:yeke_jarquchi_court_position ?= {
					has_variable = performing_improve_court
				}
			}
			add = {
				value = liege.court_position:yeke_jarquchi_court_position.yeke_jarquchi_improve_court_value
				desc = yeke_jarquchi_improve_court_aptitude_desc
			}
		}
	}

	is_shown = {
		trigger_if = {
			limit = {
				is_ai = yes
				highest_held_title_tier <= tier_county
			}
			always = no
		}
	}

	valid_position = {
		highest_held_title_tier >= tier_county
		OR = {
			has_government = administrative_government
			custom_description = {
				text = already_established_criminal_underground
				house ?= { has_house_modifier = sas_establish_criminal_underground_house_modifier }
			}
		}
	}

	is_shown_character = {
		scope:employee = {
			court_position_employee_shown_trigger = yes
		}
	}

	valid_character = {
		scope:employee = {
			femme_fatale_validity_trigger = { EMPLOYER = scope:liege }
		}
	}

	revoke_cost = {
		prestige =  {
			value = major_court_position_prestige_revoke_cost
			# Set to 0 via obligation hook
			if = {
				limit = {
					court_position_obligation_hook_salary_trigger = {
						POS = femme_fatale
						LIEGE = root
					}
				}
				multiply = 0
			}
			# Avoid AI prestige issues
			if = {
				limit = {
					exists = this
					is_ai = yes
				}
				multiply = 0
			}
			# Special case for events
			if = {
				limit = {
					exists = this
					has_character_flag = free_revoke_cost
				}
				multiply = 0
			}
			#Lowering for LAAMPS, to match Camp Officer revoke cost
			if = {
				limit = {
					exists = this
					government_has_flag = government_is_landless_adventurer
				}
				value = minor_court_position_prestige_revoke_cost
			}
		}
	}

	salary = {
		gold = {
			value = 0
			if = {
				limit = {
					scope:liege = {
						has_treasury = no
					}
				}
				add = femme_fatale_total_salary_value
			}
		}
		treasury = {
			value = 0
			if = {
				limit = {
					scope:liege = {
						has_treasury = yes
					}
				}
				add = femme_fatale_total_salary_value
			}
		}
		prestige = {
			value = femme_fatale_prestige_salary_value
		}
		round = no
	}

	received_salary = {
		gold = {
			value = femme_fatale_total_salary_value
		}
		prestige = {
			value = femme_fatale_prestige_salary_value
		}
		round = no
	}

	is_powerful_agent = yes

	scaling_employer_modifiers = {
		terrible = {
			courtier_and_guest_opinion = -5
			vassal_opinion = -2
			owned_scheme_secrecy_add = -5
		}
		poor = {
			courtier_and_guest_opinion = -2
			vassal_opinion = -1
			owned_scheme_secrecy_add = -2
		}
		average = {
			courtier_and_guest_opinion = 3
			vassal_opinion = 1
			owned_scheme_secrecy_add = 3
		}
		good = {
			courtier_and_guest_opinion = 7
			vassal_opinion = 3
			owned_scheme_secrecy_add = 7
		}
		excellent = {
			courtier_and_guest_opinion = 10
			vassal_opinion = 5
			owned_scheme_secrecy_add = 10
		}
	}

	custom_employer_modifier_description = femme_fatale_custom_effect_description_poison_opinion
	custom_employer_modifier_description = femme_fatale_custom_effect_description_fabricate_claim

	modifier = {
		intrigue = 2
		general_opinion = 10
		max_personal_schemes_add = 1
		monthly_intrigue_lifestyle_xp_gain_mult =  0.2
	}

	on_court_position_received = {
		basic_gained_court_position_effect = yes
		scope:liege = {
			set_variable = {
				name = femme_fatale_is_poisoning_opinion
				value = no
			}
			set_variable = {
				name = femme_fatale_is_embracing_the_spiders
				value = no
			}
			set_variable = {
				name = femme_fatale_is_soothing_and_surveiling
				value = no
			}
		}
	}
	on_court_position_revoked = {
		basic_revoked_court_position_effect = yes
		scope:liege = { 
			remove_variable = femme_fatale_is_poisoning_opinion
			remove_variable = femme_fatale_is_embracing_the_spiders
			remove_variable = femme_fatale_is_soothing_and_surveiling
		}
	}
	on_court_position_invalidated = {
		basic_invalidated_court_position_effect = yes
		scope:liege = { 
			remove_variable = femme_fatale_is_poisoning_opinion
			remove_variable = femme_fatale_is_embracing_the_spiders
			remove_variable = femme_fatale_is_soothing_and_surveiling
		}
	}
	on_court_position_vacated = {
		court_position_vacated_event_effect = {
			COURT_POSITION_TYPE = femme_fatale_court_position
		}
		scope:liege = { 
			remove_variable = femme_fatale_is_poisoning_opinion
			remove_variable = femme_fatale_is_embracing_the_spiders
			remove_variable = femme_fatale_is_soothing_and_surveiling
		}
	}

	# Scripted value to determine whether an AI will hire/fire a position or not. Hires over 0, fires under -50. Look in _court_positions.info for more extensive documentation!
	ai_position_score = {
		value = 50
		scope:liege = {
			# More likely to hire if a Schemer
			if = {
				limit = {
					has_trait = schemer
				}
				add = 25
			}
		}
		scope:liege = {
			if = {
				limit = {
					OR = {
						has_trait = seducer
						has_trait = ambitious
					}
				}
				add = 25
			}
		}
		# More likely to if good at diplomacy and intrigue
		scope:liege = {
			if = {
				limit = {
					diplomacy >= 10
					intrigue >= 10
				}
				add = 50
			}
		}
		add = court_position_debt_considerations_value
	}
	ai_candidate_score = {
		value = 50
		add = scope:base_value
		add = court_position_candidate_score_base_value
		add = court_position_candidate_aptitude_value
		if = { # To stop petitioned positions immediately being fired by the AI
			limit = {
				court_position_petition_ai_weight_trigger = { ROLE = femme_fatale }
			}
			add = 1000
		}
	}
}
