rabble_rouser_task_rouse_rabble = {
	court_position_types = { 
		rabble_rouser_court_position
	}
	cost = {
		round = no
		gold = {
			add = {
				value = monthly_court_position_task_cost
				multiply = 2
				desc = COURT_POSITION_TASK_COST_BREAKDOWN_BASE
				format = "BASE_VALUE_FORMAT_DECIMALS_PLUS_NEGATIVE"
			}
		}
	}

	employee_modifier = {
		monthly_prestige = minor_court_position_task_prestige_salary
	}

	scaling_employer_modifiers = {
		terrible = { negate_health_penalty_add = 0.01 }
		poor = { negate_health_penalty_add = 0.03 }
		average = { negate_health_penalty_add = 0.05 }
		good = { negate_health_penalty_add = 0.07 }
		excellent = { negate_health_penalty_add = 0.1 }
	}

	ai_will_do = {
		value = 0
		if = {
			limit = {
				OR = {
					health <= 2.0  # Poor health
					has_trait = ill
					has_trait = infirm
				}
			}
			add = 100
		}
	}
}
