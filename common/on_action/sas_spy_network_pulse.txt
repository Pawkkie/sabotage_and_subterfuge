on_game_start = {
	on_actions = {
		initialize_can_use_spy_network
	}
}

# Sets up which houses are initially allowed to use the spy network
initialize_can_use_spy_network = {
	effect = {
		every_ruler	= {
			limit = { 
				NOT = { house = { has_house_modifier = house_of_sabotage_and_subterfuge } } 
				is_house_head = yes
				has_government = administrative_government
			}
			house = {
				add_house_modifier = {
					modifier = house_of_sabotage_and_subterfuge
				}
			}
		}
	}
}

three_year_playable_pulse = {
	on_actions = {
		ai_employ_spy_pulse
		learn_innovations_pulse
	}
}

quarterly_playable_pulse = {
	on_actions = {
		gather_secrets_pulse
		learn_innovations_pulse
	}
}

initialize_can_use_spy_network = {
	
}

# This pulse triggers the spy network events that make spies discover secrets and then reveal them to their employer
gather_secrets_pulse = {
	trigger = {
		has_variable_list = spy_network_members
		is_ruler = yes
		is_alive = yes
		is_landed = yes
		any_in_list = {
			variable = spy_network_members
			count >= 1
		}
	}
	effect = {
		trigger_event = {
			id = agot_spy_network.0100
		}
	}
}

learn_innovations_pulse = {
	trigger = {
		has_variable_list = spy_network_members
		is_ruler = yes
		is_alive = yes
		is_landed = yes
		any_in_list = {
			variable = spy_network_members
			count >= 1
		}
	}
	effect = {
		trigger_event = {
			id = agot_spy_network.0105
		}
	}
}

ai_employ_spy_pulse = {
	trigger = {
		is_landed = yes
		is_ai = yes
		intrigue >= 12
		OR = {
			NOT = { has_variable_list = spy_network_members }
			any_in_list = {
				variable = spy_network_members
				is_alive = yes
				count < spy_network_limit
			}
		}
	}
	effect = {
		trigger_event = {
			id = agot_spy_network.0300
		}
	}
}
