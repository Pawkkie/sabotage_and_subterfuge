on_death = {
	on_actions = { on_death_black_book_inherit_secrets }
}

on_death_black_book_inherit_secrets = {
	trigger = {
		is_ai = no
		exists = player_heir
	}
	effect = {
		if = {
			limit = {
				any_equipped_character_artifact = {
					has_artifact_modifier = artifact_black_book_modifier
				}
				house ?= {
					OR = {
						has_house_modifier = sas_house_of_sabotage_and_subterfuge_modifier_large
						has_house_modifier = sas_house_of_sabotage_and_subterfuge_modifier_medium
						has_house_modifier = sas_house_of_sabotage_and_subterfuge_modifier_small
					}
				}
			}
		}
		black_book_inherit_secrets_effect = yes
	}
}
