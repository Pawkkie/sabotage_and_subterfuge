sas_induct_bastard_interaction = {
    icon = icon_dynasty
    category = interaction_category_friendly
    desc = sas_induct_bastard_interaction_desc

    is_shown = {
        scope:actor = {
            is_ai = no
            has_perk = enticing_opportunity_perk
        }

        scope:recipient = {
            has_trait = disputed_heritage
        }

        OR = {
            scope:recipient.real_father = scope:actor
            scope:recipient.real_father.dynasty = scope:actor.dynasty
        }

        NOT = {
            scope:actor.dynasty = scope:recipient.dynasty
        }
    }

    is_valid = {
        scope:actor.dynasty = {
            dynasty_prestige >= medium_dynasty_prestige_value
        }
    }

    auto_accept = yes

    cost = {
        renown = medium_dynasty_prestige_value
    }

    on_accept = {
        scope:actor = {
            stress_impact = {
                arrogant = minor_stress_impact_gain
            }
        }

        scope:recipient = {
            set_father = scope:recipient.real_father
            set_house = scope:recipient.real_father.house
            remove_trait = disputed_heritage
            
            if = {
                limit = {
                    faith = { has_doctrine_parameter = bastards_legitimize }
                }
                add_trait = legitimized_bastard
            }
        }

        hidden_effect = {
            every_sibling = {
                limit = {
                    is_landed = yes
                    NOT = { this = scope:actor }
                }
                trigger_event = bastard_interaction.0010
            }            
        }
    }
}

# Reveal true father
sas_reveal_true_father_interaction = {
    icon = icon_dynasty
    category = interaction_category_friendly
    desc = sas_reveal_true_father_interaction_desc

    is_shown = {
        scope:actor = {
            is_ai = no
            has_perk = enticing_opportunity_perk
            scope:recipient.mother = scope:actor
        }

        scope:recipient = {
            OR = {
                has_trait = disputed_heritage
                has_trait = bastard
                has_trait = legitimized_bastard
            }
        }

        NOT = {
            scope:recipient.real_father = scope:recipient.father
        }
    }

    is_valid = {
        scope:actor.dynasty = {
            prestige >= medium_prestige_value
        }
    }

    cost = {
        prestige = medium_prestige_value
    }

    on_accept = {
        scope:actor = {
            stress_impact = {
                arrogant = minor_stress_impact_gain
            }
        }

        scope:recipient = {
            set_father = scope:recipient.real_father
            
            if = {
                limit = {
                    has_trait = disputed_heritage
                }
                remove_trait = disputed_heritage
            }
            if = {
                limit = {
                    NOT = {
                        has_trait = bastard
                        has_trait = legitimized_bastard
                    }
                }
                add_trait = bastard
            }
            else_if = {
                limit = {
                    faith = { has_doctrine_parameter = bastards_legitimize }
                }
                add_trait = legitimized_bastard
            }
        }
    }
}
