			}
			# Nomadic story reasons
			custom_description = {
				text = provoke_target_greatest_of_khans

					NOT = {

							OR = {

	# On Actions
	on_start = {
		set_variable = {
			name = apply_countermeasures
			value = flag:indirect
		}
		add_scheme_starting_opportunities_intrigue_effect = yes
		# Fallback agents — a balanced config if you start the scheme from script.
		if = {
			limit = {
				scheme_owner = {
					NOT = { has_variable = agents_added }
				}
			}
			add_agent_slot = agent_rabble_rouser
			add_agent_slot = agent_diplomat
			add_agent_slot = agent_gabbler
			add_agent_slot = agent_herald
			add_agent_slot = agent_justiciar
		}
	}
	on_phase_completed = {
		suppress_scheme_follow_up_event_till_input_given_effect = yes
		provoke_vassals_scheme_prep_effect = yes
		sas_cap_schemes_and_fire_reminders_effect = yes
	}
	on_hud_click  = { provoke_vassals_scheme_prep_effect = yes }
	on_monthly = {
		sas_hostile_scheme_monthly_discovery_chance_effect = yes
		if = {
			limit = {
				NOT = { exists = scope:discovery_event_happening }
			}
			scheme_owner = {
				trigger_event = {
					on_action = provoke_vassals_ongoing
					days = { 1 30 }
				}
			}
		}
	}
	on_semiyearly = {
		if = {
			limit = {
				scheme_owner = { is_ai = yes }
			}
			provoke_vassals_scheme_prep_effect = yes
		}
	}
	on_invalidated = {
		scheme_target_character = {
			save_scope_as = target
		}
		scheme_owner = {
			save_scope_as = owner
		}
		# Target death
		if = {
			limit = {
				scope:target = { is_alive = no }
			}
			scope:owner = {
				send_interface_toast = {
					title = provoke_vassals_invalidated_title
					custom_tooltip = provoke_vassals_invalidated_dead
					left_icon = scope:target
				}
			}
		}
		# Target out of range
		if = {
			limit = {
				scope:target = { 
					NOT = { in_diplomatic_range = scope:owner } 
				}
			}
			scope:owner = {
				send_interface_toast = {
					title = provoke_vassals_invalidated_title
					left_icon = scope:target
					custom_description_no_bullet = {
						object = scope:target
						text = scheme_target_not_in_diplomatic_range
					}
				}
			}
		}
		# Target no longer controls at least Kingdom
		if = {
			limit = {
				scope:target = {
					NOT = { highest_held_title_tier >= tier_kingdom }
				}
			}
			scope:owner = {
				send_interface_toast = {
					title = provoke_vassals_invalidated_title
					custom_tooltip = provoke_vassals_invalidated_target_lost_title
					left_icon = scope:target
				}
			}
		}

					any_targeting_faction = {

					}
				}
			}

				send_interface_toast = {
					title = provoke_vassals_invalidated_title
					custom_tooltip = provoke_vassals_invalidated_has_independence_faction
					left_icon = scope:target
				}
			}
		}

		# Target not already dealing with dissolution faction
		if = {
			limit = {
				scope:target = {
					any_targeting_faction = {
						faction_is_type = nation_fracturing_faction
					}
				}
			}
			scope:owner = {
				send_interface_toast = {
					title = provoke_vassals_invalidated_title
					custom_tooltip = provoke_vassals_invalidated_has_dissolution_faction
					left_icon = scope:target
				}
			}
		}

		# Has any vassal that can create a faction

			limit = {
				NOT = {
					scope:target = {
						any_vassal = {
							is_landed = yes
							AND = {
								is_at_war = no
								NOT = {
									AND = {
										has_opinion_modifier = {
											modifier = became_vassal
											target = scope:target
										}

										type = independence_faction

									}
									can_create_faction = {
										type = nation_fracturing_faction
										target = scope:target
									}
								}
							}
						}
					}
				}
			}
			scope:owner = {
				send_interface_toast = {
					title = provoke_vassals_invalidated_title
					custom_tooltip = provoke_vassals_invalidated_no_eligible_vassals
					left_icon = scope:target
				}
			}
		}

		# Recent independence faction revolt
		if = {
			limit = {
				scope:target = {
					has_character_flag = recent_independence_faction_war
				}
			}
			scope:owner = {
				send_interface_toast = {

						government_allows = administrative

					title = provoke_vassals_invalidated_title

				}
			}
		}

		# Nomadic and tribal confederate partition succession
		if = {
			limit = {
				scope:target = {
					AND = {
						OR = {
							government_has_flag = government_is_nomadic
							government_has_flag = government_is_tribal
						}
						primary_title = {
							has_title_law = confederation_elective_succession_law
						}
					}
				}
			}
			scope:owner = {
				send_interface_toast = {
					title = provoke_vassals_invalidated_title
					custom_tooltip = provoke_vassals_invalidated_confederate_partition
				}
			}
		}
			
		# Nomadic story reasons
		if = {
			limit = {
				scope:target = {
					any_owned_story = {

							story_type = story_greatest_of_khans

					}

					}

		if = {
			limit = {

			}
			scope:owner = {
				send_interface_toast = {
					title = provoke_vassals_invalidated_title
					left_icon = scope:owner.court_position:rabble_rouser_court_position

						subject = scope:owner.court_position:rabble_rouser_court_position
						text = provoke_vassals_invalidated_not_provoking_vassals
					}
				}
			}
