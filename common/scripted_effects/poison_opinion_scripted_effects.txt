poison_opinion_end_effect = {
	if = {
		limit = {
			scope:target = {
				NOT = {
					has_opinion_modifier = {
						target = scope:owner
						modifier = scheme_poison_opinion_opinion
						value <= poison_opinion_min_value
					}
				}
			}
		}
		if = {
			limit = {
				exists = scope:scheme
			}
			scope:scheme = {
				reset_failed_scheme_effect = yes
			}
		}
		if = {
			limit = {
				exists = scope:scheme_successful
			}
			custom_tooltip = poison_opinion_continue
		}
		else = {
			custom_tooltip = poison_opinion_try_again
		}
		
		
	}
	else = {
		scope:owner = {
			send_interface_toast = {
				title = poison_opinion_complete
				left_icon = scope:target
				scope:scheme = {
					end_scheme = yes
				}
			}
		}
	}
}

poison_opinion_ongoing_1002_outcome_effect = {
	if  = {
		limit = {
			exists = scope:believer
			has_relation_friend = scope:believer
		}
		show_as_tooltip = { theology_3021_a_success_effect = yes }
	}
	else = {
		random_list = {
			1 = {
				desc = diplomacy_majesty.0004.success.tt
				show_chance = no
				show_as_tooltip = {
					scope:compliment_receiver = {
						add_opinion = {
							target = root
							modifier = respect_opinion
							opinion = 10
						}
					}
				}
			}
			1 = {
				desc = diplomacy_majesty.0004.failure.tt
				show_chance = no
				show_as_tooltip = {
					scope:compliment_receiver = {
						add_opinion = {
							target = root
							modifier = respect_opinion
							opinion = -5
						}
					}
				}
			}
		}
	}

	hidden_effect = {
		if = {
			limit = {
				exists = scope:compliment_outcome
				scope:compliment_outcome = flag:good
			}
			trigger_event = poison_opinion_ongoing.1003
		}
		else_if = { #Is treated as a good outcome here
			limit = {
				exists = scope:compliment_outcome
				scope:compliment_outcome = flag:neutral
			}
			trigger_event = poison_opinion_ongoing.1003
		}
		else_if = {
			limit = {
				exists = scope:compliment_outcome
				scope:compliment_outcome = flag:bad
			}
			trigger_event = poison_opinion_ongoing.1004
		}
	}
}
